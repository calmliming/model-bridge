接入示例
# 设置好 MODEL_BRIDGE_API_KEY 后运行
curl http://localhost:3000/v1/models \
  -H "Authorization: Bearer $MODEL_BRIDGE_API_KEY"
// 在服务端调用，密钥从环境变量读取
const response = await fetch(
  'http://localhost:3000/v1/models',
  {
    headers: {
      Authorization:
        `Bearer ${process.env.MODEL_BRIDGE_API_KEY}`
    }
  }
);
const models = await response.json();
console.log(models);
# 在服务端调用，密钥从环境变量读取
import os
import json
from urllib.request import Request, urlopen

request = Request(
    "http://localhost:3000/v1/models",
    headers={"Authorization": "Bearer " +
             os.environ["MODEL_BRIDGE_API_KEY"]}
)
with urlopen(request) as response:
    print(json.load(response))
